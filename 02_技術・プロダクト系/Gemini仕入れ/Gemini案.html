    <script>

        // Init Date

        document.getElementById('today').valueAsDate = new Date();

        document.getElementById('currentDateDisplay').innerText = new Date().toLocaleDateString('ja-JP', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });


        let chartInstance = null;


        function calculate() {

            // Get values

            const today = new Date(document.getElementById('today').value);

            const targetDate = new Date(document.getElementById('targetDate').value);

            const currentInv = parseInt(document.getElementById('currentInventory').value);

            const targetInv = parseInt(document.getElementById('targetInventory').value);

            const dailyOutput = parseInt(document.getElementById('dailyOutput').value);


            // Calc Time

            const diffTime = targetDate - today;

            const daysRemaining = Math.ceil(diffTime / (1000 * 60 * 60 * 24)); 


            // Calc Inventory

            const gap = currentInv - targetInv;

            

            // Logic Check

            if (daysRemaining <= 0) {

                document.getElementById('daysRemaining').innerText = "0";

                document.getElementById('maxInputLimit').innerText = "到達";

                return;

            }


            const dailyBurn = Math.ceil(gap / daysRemaining);

            const maxInput = dailyOutput - dailyBurn;


            // Update DOM

            document.getElementById('daysRemaining').innerText = daysRemaining + "日";

            document.getElementById('inventoryGap').innerText = gap.toLocaleString() + "冊";

            document.getElementById('dailyBurn').innerText = "▼" + dailyBurn.toLocaleString();

            

            const limitEl = document.getElementById('maxInputLimit');

            limitEl.innerText = maxInput.toLocaleString();

            

            // Color Logic

            if (maxInput < 15000) {

                limitEl.className = "text-6xl font-bold tracking-tighter text-red-500"; // Too tight

            } else {

                limitEl.className = "text-6xl font-bold tracking-tighter text-yellow-400"; // OK

            }


            document.getElementById('calcOutput').innerText = dailyOutput.toLocaleString();

            document.getElementById('calcBurn').innerText = dailyBurn.toLocaleString();


            updateChart(today, daysRemaining, currentInv, targetInv);

        }


        function updateChart(startDate, days, startVal, endVal) {

            const ctx = document.getElementById('burnDownChart').getContext('2d');

            

            // Generate Labels and Data

            const labels = [];

            const dataIdeal = [];

            const step = (startVal - endVal) / days;


            for (let i = 0; i <= days; i++) {

                const d = new Date(startDate);

                d.setDate(d.getDate() + i);

                labels.push(`${d.getMonth()+1}/${d.getDate()}`);

                

                let val = startVal - (step * i);

                dataIdeal.push(val);

            }


            if (chartInstance) {

                chartInstance.destroy();

            }


            chartInstance = new Chart(ctx, {

                type: 'line',

                data: {

                    labels: labels,

                    datasets: [{

                        label: '理想的な在庫推移 (Burndown)',

                        data: dataIdeal,

                        borderColor: '#3b82f6',

                        backgroundColor: 'rgba(59, 130, 246, 0.1)',

                        borderWidth: 2,

                        fill: true,

                        pointRadius: 0,

                        tension: 0.1

                    }]

                },

                options: {

                    responsive: true,

                    maintainAspectRatio: false,

                    plugins: {

                        legend: { display: false },

                        tooltip: { mode: 'index', intersect: false }

                    },

                    scales: {

                        y: {

                            beginAtZero: false,

                            min: 0,

                            max: Math.max(startVal * 1.1)

                        }

                    }

                }

            });

        }


        // Initial Run

        calculate();
